  <script>

    // PASTE YOUR APPS SCRIPT WEB APP URL BELOW:

    const API_URL = "PASTE_YOUR_WEBAPP_URL_HERE";


    let studentId = localStorage.getItem("oge_student_id") || "";

    const chatBox = document.getElementById("chat-box");

    const inputField = document.getElementById("user-input");

    const continueBar = document.getElementById("continue-bar");


    if (studentId) {

      document.getElementById("login-overlay").style.display = "none";

      document.getElementById("header-title").innerText = "🎓 " + studentId;

      sendAction("check");

    }


    function saveStudentId() {

      const val = document.getElementById("student-id-input").value.trim();

      if (!val) return;

      studentId = val;

      localStorage.setItem("oge_student_id", studentId);

      document.getElementById("login-overlay").style.display = "none";

      document.getElementById("header-title").innerText = "🎓 " + studentId;

      sendAction("check");

    }


    function appendMessage(htmlContent, sender) {

      const div = document.createElement("div");

      div.className = "msg " + sender;

      if (sender === "bot") {

        div.innerHTML = htmlContent;

      } else {

        div.textContent = htmlContent;

      }

      chatBox.appendChild(div);

      chatBox.scrollTop = chatBox.scrollHeight;

    }


    async function callBackend(payload) {

      const typingDiv = document.createElement("div");

      typingDiv.className = "msg bot";

      typingDiv.textContent = "⏳...";

      chatBox.appendChild(typingDiv);

      chatBox.scrollTop = chatBox.scrollHeight;


      try {

        // Sending as text/plain avoids CORS preflight issues with Google Apps Script

        const response = await fetch(API_URL, {

          method: "POST",

          headers: { "Content-Type": "text/plain;charset=utf-8" },

          body: JSON.stringify(payload)

        });

        const data = await response.json();

        chatBox.removeChild(typingDiv);


        if (data.messages && data.messages.length > 0) {

          data.messages.forEach(m => appendMessage(m, "bot"));

        }

        continueBar.style.display = data.isPaused ? "block" : "none";

      } catch (err) {

        chatBox.removeChild(typingDiv);

        appendMessage("⚠️ Ошибка соединения с сервером. Проверь интернет.", "bot");

      }

    }


    function sendText() {

      const text = inputField.value.trim();

      if (!text) return;

      inputField.value = "";

      appendMessage(text, "user");

      callBackend({ student_id: studentId, action: "message", text: text });

    }


    function sendAction(actionType) {

      if (actionType === "start") appendMessage("🚀 Начать тренировку", "user");

      if (actionType === "pause") appendMessage("⏸ Пауза", "user");

      if (actionType === "continue") appendMessage("▶️ Продолжить", "user");

      callBackend({ student_id: studentId, action: actionType, text: "" });

    }


    inputField.addEventListener("keydown", function(e) {

      if (e.key === "Enter") sendText();

    });
